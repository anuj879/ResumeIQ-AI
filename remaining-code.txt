function useRoute() {
  const get = () => (location.hash || "#/").replace(/^#\/?/, "");

  const [route, setRoute] = useState(get);

  useEffect(() => {
    const handleRoute = () => {
      setRoute(get());
      window.scrollTo(0, 0);
    };

    window.addEventListener("hashchange", handleRoute);
    return () => window.removeEventListener("hashchange", handleRoute);
  }, []);

  return route;
}

          </p>

      {/* Quick Access */}
      <section>

          <p className="mt-1 text-sm text-white/45">
            Access your career tools in one place.
          </p>
        </div>

        <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
          {quickLinks.map((tab) => {
            const Icon = tab.icon;

            return (
              <a
                key={tab.id}
                href={`#/app/${tab.id}`}
                className="group flex items-center gap-3 rounded-xl border border-white/10 bg-white/[0.035] p-4 transition duration-300 hover:-translate-y-1 hover:border-violet-400/30 hover:bg-violet-500/[0.08]"
              >
                <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-violet-500/10 text-violet-300 transition group-hover:bg-violet-500/20">
                  <Icon size={19} />
                </div>

                <div className="min-w-0 flex-1">
                  <p className="text-sm font-semibold">{tab.label}</p>
                  <p className="mt-1 text-xs text-white/40">
                    Open workspace
                  </p>
                </div>

                <ArrowUpRight
                  size={17}
                  className="text-white/30 transition group-hover:text-violet-300"
                />
              </a>
            );
          })}
        </div>
      </section>

      {/* Recent Activity */}
      <section className="rounded-2xl border border-white/10 bg-white/[0.035] p-5">
        <h2 className="font-bold">Recent Activity</h2>

        {done.length ? (
          <div className="mt-4 space-y-3">
            {done.map((id) => (
              <div
                key={id}
                className="flex items-center justify-between border-b border-white/[0.06] pb-3 last:border-0 last:pb-0"
              >
                <span className="text-sm text-white/70">
                  {TABS.find((tab) => tab.id === id)?.label || id}
                </span>

                <span className="text-xs text-emerald-300">Completed</span>
              </div>
            ))}
          </div>
        ) : (
          <p className="mt-3 text-sm text-white/40">
            No activity yet. Your completed analyses will appear here.
          </p>
        )}
      </section>
    </div>
  );
}


========== frontend/src/api.js ==========

const API_BASE = (import.meta.env.VITE_API_URL || "").replace(/\/$/, "");
const apiUrl = (path) => `${API_BASE}${path}`;

const NET = "Cannot reach the server. Please try again.";

async function handle(r) {
  const d = await r.json().catch(() => ({}));
  if (!r.ok) throw new Error(d.error || "Request failed.");
  return d;
}

export const extractResume = (file, onProgress) =>
  new Promise((resolve, reject) => {
    const form = new FormData();
    form.append("resume", file);

    const x = new XMLHttpRequest();
    x.open("POST", apiUrl("/api/extract"));

    x.upload.onprogress = (e) =>
      e.lengthComputable &&
      onProgress?.(Math.round((e.loaded / e.total) * 100));

    x.onload = () => {
      let d = {};
      try {
        d = JSON.parse(x.responseText);
      } catch {
        /* ignore */
      }

      x.status >= 200 && x.status < 300
        ? resolve(d)
        : reject(new Error(d.error || "Upload failed."));
    };

    x.onerror = () => reject(new Error(NET));
    x.send(form);
  });

export const runTask = (task, body) =>
  fetch(apiUrl("/api/analyze"), {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ task, ...body }),
  })
    .catch(() => {
      throw new Error(NET);
    })
    .then(handle)
    .then((d) => d.result);

export const health = () => fetch(apiUrl("/api/health")).then(handle);
